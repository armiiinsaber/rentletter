## Part 1. The scoring model

### 1a. What the model is

`lib/scoring.js` (96 lines) and `lib/scorecard.js` (88 lines) are reproduced here in full because every finding below cites them.

```js
// lib/scoring.js
function round1(n) { return Math.round(n * 10) / 10; }
function round2(n) { return Math.round(n * 100) / 100; }
function clamp(n, lo, hi) { return Math.max(lo, Math.min(hi, n)); }
function interp(x, pts) { /* linear interpolation across [x, y] anchors, flat beyond the ends */ }

export function incomeLevelScore(annualHouseholdIncome) {          // :36
  const x = Math.max(0, Number(annualHouseholdIncome) || 0);
  if (x <= 0) return 2.5;
  const MID = 56000, WIDTH = 16000;
  return round2(5 / (1 + Math.exp(-(x - MID) / WIDTH)));
}
export function rentToIncomeScore(ratioPct) {                      // :49
  const r = Number(ratioPct);
  if (!isFinite(r) || r <= 0) return 3;
  return round2(interp(r, [[0, 5.0], [30, 5.0], [39, 4.7], [50, 3.8], [60, 2.4], [70, 1.2], [100, 1.0]]));
}
export function rentAffordabilityScore(annualHouseholdIncome, ratioPct) {   // :61
  return round1(clamp(0.6 * rentToIncomeScore(ratioPct) + 0.4 * incomeLevelScore(annualHouseholdIncome), 1, 5));
}
export function employmentStabilityScore(yearsAtJob) {             // :70
  return round1(interp(Math.max(0, Number(yearsAtJob) || 0), [[0, 2.5], [0.5, 3.0], [1, 3.7], [2, 4.4], [3, 4.8], [4, 5.0]]));
}
export function rentalHistoryScore({ yearsAtPrevious, previousLandlordName, previousAddress, referencesCount = 0 }) {   // :80
  const histYears = Math.max(0, Number(yearsAtPrevious) || 0);
  let base;
  if (histYears >= 2 && previousLandlordName) base = 4.5;
  else if (histYears >= 1 && previousLandlordName) base = 3.8;
  else if (previousAddress) base = 3.0;
  else base = 3.0;
  const refs = Math.max(0, Number(referencesCount) || 0);
  const refBump = refs >= 2 ? 0.7 : refs === 1 ? 0.35 : 0;
  return round1(clamp(base + refBump, 1, 5));
}
export function overallScore(scores) {                             // :93
  const { incomeStability, rentAffordability, rentalHistory, longTermIntent, disclosures } = scores;
  return round1((incomeStability + rentAffordability + rentalHistory + longTermIntent + disclosures) / 5);
}
```

```js
// lib/scorecard.js
export function calculateScorecard(data) {                          // :13
  const { yearsAtJob, householdAnnualIncome, householdRentToIncomeRatio, hasCoApplicant,
    previousAddress, yearsAtPrevious, previousLandlordName, referencesCount, reasonForMoving, redFlags } = data;

  const jobYears = parseFloat(yearsAtJob) || 0;                     // :21
  const incomeStability = employmentStabilityScore(jobYears);

  let rentAffordability = 3;                                        // :31
  if (householdRentToIncomeRatio !== null && householdRentToIncomeRatio !== undefined) {
    rentAffordability = rentAffordabilityScore(householdAnnualIncome, householdRentToIncomeRatio);
  }

  const histYears = parseFloat(yearsAtPrevious) || 0;               // :39
  const rentalHistory = rentalHistoryScore({ yearsAtPrevious: histYears, previousLandlordName, previousAddress, referencesCount });

  let longTermIntent = 4;                                           // :50
  const reason = (reasonForMoving || '').toLowerCase();
  const strongIntentKeywords = ['new job', 'job', 'school', 'university', 'partner', 'family', 'closer to work', 'commute', 'permanent', 'long-term', 'settle'];
  const shortIntentKeywords = ['temporary', 'short-term', 'few months', 'travel'];
  if (strongIntentKeywords.some(k => reason.includes(k))) longTermIntent = 5;
  else if (shortIntentKeywords.some(k => reason.includes(k))) longTermIntent = 3;

  let disclosures = 5;                                              // :62
  if (redFlags && redFlags.trim().length > 0) {
    const flagText = redFlags.toLowerCase();
    if (flagText.includes('bankruptcy') || flagText.includes('eviction')) disclosures = 3;
    else if (flagText.includes('credit') || flagText.includes('gap')) disclosures = 4;
    else disclosures = 4;
  }

  return { incomeStability: {...}, rentAffordability: {...}, rentalHistory: {...}, longTermIntent: {...}, disclosures: {...},
    overall: overallScore({ incomeStability, rentAffordability, rentalHistory, longTermIntent, disclosures }) };   // :78-87
}
```

Component by component:

| Component (weight 1/5 each) | Measures | Reads | Stated or verified | Screenable under OHRC / BC Code |
|---|---|---|---|---|
| `incomeStability` (`scoring.js:70`) | tenure at current job | `years_at_job` (form field) | stated | Yes. Employment history is permitted under Ontario Reg 290/98 s.1 and BC practice. Note the name is misleading: it measures tenure, not income. |
| `rentAffordability` (`scoring.js:61`) | 60% rent share, 40% absolute income level | `rent_to_income_ratio` (now derived from the live listing rent since `014ef1d`, `lib/deriveScorecard.js:22-29`), `annual_income`, `co_applicant.annualIncome` | stated (document verification never enters, see 1b iii) | Rent share and income information are permitted, but only alongside other information and not as a rigid cutoff (Reg 290/98 s.1(1); *Shelter Corp. v. Ontario (Human Rights Commission)* found rigid rent to income ratios discriminatory). The 40% `incomeLevelScore` term rewards **absolute income beyond what the unit costs** (`scoring.js:36-43`): two applicants who both clear the rent at 25% are ranked apart by how rich they are. `GUESS`: that is the part a tribunal would look at hardest, because it is not an affordability test for this unit. |
| `rentalHistory` (`scoring.js:80`) | prior tenancy length, landlord reference present, references count | `years_at_previous`, `prev_landlord_name`, `prev_address`, `references.length` | stated (presence of a name, never an outcome) | Yes. Rental history and references are permitted. First time renters are held neutral at 3.0 (`:85-86`), which is correct. |
| `longTermIntent` (`scorecard.js:50-59`) | keyword match on free text "Why are you moving?" | `reason_for_moving` | stated free text | **No.** See below. |
| `disclosures` (`scorecard.js:62-76`) | keyword match on free text "Anything to address?" | `redFlags` | stated free text | Partly. See below. |

**`longTermIntent`, in detail.** The keyword list at `scorecard.js:52` awards 5 (a full point above the default 4) when the tenant's reason contains `family`, `partner`, `school`, or `university`. Verified with the list itself: "moving closer to family" scores 5; "moving for work" scores 4 (the word `work` alone is not in the list; only `closer to work` is). "my partner and I are moving in together" scores 5. "starting university in September" scores 5. "need a bigger place for the baby" scores 4.

- `family` is a direct proxy for **family status** (Ontario Code s.2(1); BC Code s.10). It **earns** a point.
- `partner` is a direct proxy for **marital status**. It earns a point.
- `school` and `university` are proxies for **age** (and, for "starting university", for being young). They earn a point.
- The short list (`:53`) docks a point for `temporary`, `short-term`, `few months`, `travel`: contract workers and people on work permits are the population that writes those words. `GUESS`: this is a proxy for **place of origin / citizenship** in practice.

The component is invisible to the realtor: `reason_for_moving` is not rendered anywhere on the listing page, the compare view, the PDF or the report text (`rg -n "reason_for_moving|redFlags|disclosures" components/dashboard/ListingView.js components/dashboard/CompareTenants.js lib/landlordReportPdf.js pages/api/listings/report-text.js`: NOT FOUND). So a fifth of the ranking moves on words the realtor never sees, several of which name protected grounds. The effect size is 0.2 on `overall` (one point over five components), which is the same magnitude as the entire affordability swing between a 33% and a 52% rent share in the current model. **Severity: high. This is the single compliance finding in the repository.**

**`disclosures`, in detail.** Any text at all in "Anything to address?" costs at least one point (`:72-74`: the `else` branch is 4, so "nothing to declare" scores 4 while an empty box scores 5). `bankruptcy` and `eviction` cost two. `credit` and `gap` cost one. Bankruptcy and credit history are financial facts and permitted. `gap` is not a financial fact: verified with the list, "a gap in employment while on maternity leave" scores 4, and parental leave, illness and caregiving are the common causes of an employment gap, all of which touch **family status, sex, and disability**. And the base penalty for having written anything punishes candour, which the form explicitly invites ("Anything to address? (gaps in history, credit, etc.)", `pages/apply/[token].js:701`). **Severity: medium high.** Also invisible to the realtor.

Held constant across the pairs (a typical stored set): `incomeStability` 4.8 (3 years at job), `rentalHistory` 4.9 (2 years prior with a landlord reference and one reference), `longTermIntent` 4, `disclosures` 5.

| i | 170,000 at rent 1,000 (7%) | 5.0 | **4.7** | | |
| ii | 60,000 at rent 2,000 (40%) | 3.9 | **4.5** | **Yes.** 40% is at the edge; 52% is over half of take home. | **No.** A 0.1 difference reads as a tie. |
| ii | 60,000 at rent 2,600 (52%) | 3.2 | **4.4** | | |

### 1c. Replacement model: **Fit**

**What the number means, in one sentence a realtor can say to a landlord:** "Out of 5, how well this applicant's income and record fit this unit's rent and your criteria, and whether we verified it."

**Design principle.** The number is about *this unit* and *this landlord's stated criteria*, it says whether its inputs were verified, and it reads nothing a tenant wrote in a free text box. Everything else the realtor needs is shown as a named fact, not folded into the number.

**Should a single number exist at all?** The case against: at one to three applicants a number invites false precision, the tick meter is the product's visual signature so it will be trusted, and any composite hides the one fact that matters most (is the income real). The case for: the realtor sends a ranked list to a landlord who will not read six facts per person, the product's positioning ("ranked, best fit first") and the film, the mockups and the report all depend on an ordinal, and a number that is honest about its inputs is more defensible than a realtor's gut ordering. **Decision: keep one number, redefine it as Fit, and make it carry a verification label ("4.6 verified" or "4.6 stated") so the two inputs are never confused.** The named facts (rent share, verification, criteria met) sit beside it on the card and in the report, so the number is never the only thing shown.

**Components and weights.**

| Component | Weight | Why this weight |
|---|---|---|
| A. Affordability against this unit | 0.5 | It is the one fact every landlord asks first, it is the fact the listing's rent actually changes, and it is the fact the realtor's criteria most often name. Half, not more, because an affordable applicant with nothing verified and no record is still a risk. |
| E. Evidence | 0.3 | Verification is the product's differentiator and the thing a landlord cannot get from a form. Nearly a third so that a verified applicant always ranks above an otherwise identical stated one by a visible margin (0.9 in the cases below). |
| R. Record | 0.2 | Tenure and references are permitted, useful, and stated; they break ties without dominating. |

**How verification enters.** Two ways. First, E is its own component, scored from the active document report (`lib/applicantSynthesis.js:25-33` already computes `incomeVerified`, `employmentVerified`, `documents`). Second, the income figure used in A is the verified figure when documents matched it (`comparisons[].found` where `status === 'match'`, the same read `lib/listingReportData.js:43-56` performs), else the stated figure, and the label on the number says which. A stated and a verified fact are different inputs and the score both ranks them apart and says so.

**How the realtor's `pref_` criteria enter.** Four of the columns written by `components/listings/ListingSetupModal.js` become inputs, so the copy "ranked against your stated criteria" (`components/dashboard/ListingView.js:787`, `lib/landlordReportPdf.js:311`) becomes true:

- `pref_rent_to_income_max_pct` sets the affordability threshold T (default 40 when unset).
- `pref_min_annual_income`: household income below it caps A at 2.0 (a named miss, not a hidden zero).
- `pref_min_years_at_job`: meeting it earns the tenure point in R.
- `pref_requires_landlord_reference`: required and absent caps R at 2.0.
- `pref_requires_employer_verification`: required and not verified caps E at 2.0.

Every cap is also rendered as a named miss on the card ("Below your $60k minimum", "No landlord reference, which you require"), so the realtor sees why. `pref_max_occupants`, `pref_smoking_allowed`, pets and move in dates never touch the number; occupancy and smoking stay as displayed facts for the realtor's own judgement (and see the compliance note on occupants in Part 6).

**The exact formula.**

```
ratio  = round(100 * rent / round(householdIncome / 12))           householdIncome = verified income if matched, else stated gross plus co applicant gross
T      = pref_rent_to_income_max_pct || 40
x      = ratio / T
A      = interp(x, [[0, 5], [0.75, 5], [1.0, 4.0], [1.25, 2.5], [1.5, 1.5], [2.0, 1.0]])
         if pref_min_annual_income and householdIncome < pref_min_annual_income: A = min(A, 2.0)
         if no rent or no income: A = null and Fit is not shown (the card shows "Rent share unknown")

E      = 5.0  income and employer both matched by documents
         4.5  income matched
         2.5  documents on file, name matches, nothing matched
         2.0  no documents
         1.0  document name mismatch or unclear
         if pref_requires_employer_verification and employer not matched: E = min(E, 2.0)

R      = clamp(2.5 + (landlord reference on file ? 1.5 : 0) + 0.5 * min(references, 2) + (years_at_job >= (pref_min_years_at_job || 1) ? 0.5 : 0), 1, 5)
         if pref_requires_landlord_reference and none on file: R = min(R, 2.0)

Fit    = round1(0.5 * A + 0.3 * E + 0.2 * R)      label: "verified" when E >= 4.5, else "stated"
```

**What the landlord report should freeze at send time.** Per applicant: name, rank, Fit and its label, A, E, R, the rent share, the household income used and whether it was verified, the criteria met and missed, the set aside reason if any; per report: the listing rent, every `pref_` value used, the model version string (`fit-v2`), the realtor's signing name, and the send timestamp. Today nothing is frozen: the PDF is built live in `pages/api/listings/send-report.js:59-62` and the only durable trace is an event with a count (`:97`). The `events.payload` cleaner (`lib/events.js:8-17`) drops nested objects and truncates strings to 200 characters, so it cannot hold a snapshot as written; a `report_snapshots` table (reviewed migration) is the right home, with the PDF bytes not stored (they are reproducible from the snapshot).

**Delete outright from the current model:**
- `longTermIntent` and both keyword lists (`scorecard.js:49-59`).
- `disclosures` and its keyword matching (`scorecard.js:61-76`). Keep the tenant's answer as a private note visible to the realtor if you want it seen at all; today it is scored and never shown.
- `incomeLevelScore` as a component of affordability (`scoring.js:36-43`, weighted at `:62`). Absolute income beyond the unit's rent is not a fit fact.
- The rent regex in `pages/api/generate.js:359` and `lib/tenantProfile.js:183` as inputs to anything (kept only as a display fallback until `estimated_rent` is written from the listing directly).
- Equal weighting in `overallScore` (`scoring.js:93-96`).

## Part 2. Ranking and utility at low volume

What the code delivers at **n = 1**, in order of what the realtor actually gets:

1. **Collection.** One invite link (`pages/api/listings/invite.js`), a standardized application instead of a PDF and three texts (`pages/apply/[token].js`), mirrored into the dashboard (`pages/api/applications/mirror.js:54`).
2. **Verification.** Documents analyzed and discarded, income and employer matched against what was stated, a name gate (`pages/api/applicants/analyze-documents.js:80`, `lib/applicantAnalysis.js:107-114`), and the tenant self upload path (`pages/api/upload/finalize.js`). This is the value SingleKey cannot copy for a realtor's workflow and it is the same at n = 1 as at n = 10.
3. **The landlord report.** `send-report.js:46` sends with one applicant. `lib/landlordReportPdf.js:311` labels it "The 1 best fit for your stated criteria", which reads oddly for one person.
4. **The paper trail.** Set aside with a screenable reason (`ListingView.js:330-345`, `lib/setAsideReasons.js`), the timeline (`lib/events.js`), and the synthesis line (`lib/applicantSynthesis.js`) that says in one line what is and is not verified.

At **n = 2** add `CompareTenants` (`ListingView.js:792`, gated on `active.length >= 2`) and the two synthesis lines side by side.

What does not deliver at low volume, with the lines:

- "Your top 5 are highlighted" and the `TOP 5` pill on every card (`ListingView.js:787`, `:808` `top5: idx < 5`) when there are two applicants.
- "Below your top 5" divider (`:801`) never appears below five, but the copy above promises it.
- The score itself: two applicants at 4.7 and 4.5 tell the realtor nothing they did not know.
- The report's "TOP MATCHES" section header for one person (`landlordReportPdf.js:311`).
- The dashboard's empty state promises "ranked" (`HomeView.js:265`, "their applications land here") and the listing page promises "ranked against your stated criteria" (`ListingView.js:779, 787`), which is not yet true (Part 1).

What should change so the product is obviously useful at one and two:

- **Listing page, n below 5:** drop `TOP 5` pills and the "top 5" sentence; replace the intro at `:787` with "Everyone who applied, checked against your criteria. Verify documents before anything goes to the landlord." At n = 1 the primary action on the card is **Request documents** (already exists, `ApplicantDocRequest`), not Set aside.
- **The card at n below 5:** show the criteria checklist (met, missed, unverified) as the second row instead of relying on the number; the number stays but the checklist is the story.
- **Compare at n = 2:** open by default when both are active (today it is a button, `:792`), with the two verification states in the first row.
- **The report at n = 1 and 2:** rename "TOP MATCHES" to "APPLICANT" or "APPLICANTS" (`landlordReportPdf.js:311`), lead with the verification result and the criteria checklist, and put Fit second. The subtitle "1 applicant, ranked best fit first" (`:230`) becomes "1 applicant, checked against your criteria".
- **Dashboard:** the listing card (`HomeView.js:334-350`) shows rent and invite state only; add one line per listing: "2 applicants, 1 verified, none sent" from the signals already loaded (`signals.applicantsByListing`), so the home page answers "what do I do next" without opening the listing.
- **Copy everywhere:** "ranked" implies a crowd; "checked" is true at any n. Keep "ranked" only in the list header when n is 5 or more.

### 4a. The dashboard load, 12 listings

`pages/landlord.js getServerSideProps` (`:37-90`) then `lib/dashboardSignals.js loadSignals` (`:12-38`):

Sequential, before signals: `auth.getUser()` (1 auth call), then `readProfile` and the listings select in parallel (2, `landlord.js:40-43`), then possibly a province update (`:59-61`) and a promo redemption plus profile re read (`:66-68`). Then `loadSignals`, whose `Promise.all` (`dashboardSignals.js:19-25`) fans out:

- latest event: 1 query
- `notificationsFor` (`lib/notificationsFeed.js`): up to 6 queries, **sequential** (`:16`, `:24`, `:30` with a possible retry at `:33`, `:39`, `:57`, `:64`)
- `inboxFor` (`lib/referrals.js:203-226`): KV `lrange` + one KV `get` per referral (`:71-75`), plus a **write** per unclaimed referral (`:207`, `saveReferral` = KV set + `referrals` upsert)
- `listFromRealtor`: KV `lrange` + one KV `get` per referral
- per listing, 12 chains in parallel, each **3 sequential** Supabase queries: junction select with `applications(*)` (`supabaseBridge.js:60`; 5 if the optional column probe fires), the listing rent select added by `014ef1d` (`:74`), then `attachDocVerifications` re selecting `doc_verifications` and `ai_insight` from the same junction rows (`:126-127`)

Count for 12 listings, happy path: 3 (auth, profile, listings) + 1 (event) + 6 (feed) + 36 (listings) = **46 Supabase round trips** plus the KV calls, on every dashboard view, before the page can paint (it is all inside `getServerSideProps`). The longest path is the feed's six sequential hops or a listing chain of three; they run side by side, so the wall time is about six sequential round trips plus payload, and the payload includes every application body (`applications(*)`) for up to 12 listings, most of which the home page reduces to counts and three cards.

**One change:** make `fetchListingApplicants` accept the listing row (every caller already has it: `dashboardSignals.js:26` has `ls`, `[id].js:54` has `listing`, `listingReportData.js:150-155` has `listing`) and select `doc_verifications` and `ai_insight` in the junction query it already runs, so `attachDocVerifications` only maps and the extra rent select disappears. Per listing: 1 query instead of 3; 12 instead of 36. **Tradeoff:** the two key attribution in `attachDocVerifications` (`:133-137`) must be preserved in the merged mapping, and `pages/api/listings/applicants.js:24-28` selects only `id` today and would need `monthly_rent` and the `pref_*` columns; a caller that passes nothing falls back to the current select.

### 4d. Work the user feels

- **The bell loads the whole dashboard on every non home page.** `components/dashboard/AssistantBell.js:26` fetches `/api/assistant/signals`, which runs `loadSignals` (`pages/api/assistant/signals.js:14`), the same 40 plus queries as the home page, on the listing page, the profile page and the billing page, to draw a badge count.
